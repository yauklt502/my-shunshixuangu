{=========================================================
  主板 · 非ST · 昨日涨停 · 竞价涨停取反 · 连板优化 v2
  用法：9:25 集合竞价结束后，条件选股 → 公式选股。
  选出后按「综合关注度」或「量占自由」降序，取前3~5。
  盘前 9:15~9:25 走势请用 scripts/preopen_pick.py。
  研究用，不构成投资建议。
=========================================================}

主板:=(CODELIKE('60') AND NOT CODELIKE('68')) OR (CODELIKE('00') AND NOT CODELIKE('30'));
非ST:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('S*ST') OR NAMELIKE('SST'));

涨停幅:=0.10;
昨涨停价:=ZTPRICE(REF(C,2),涨停幅);
昨日涨停:=REF(C,1)>=昨涨停价;

今涨停价:=ZTPRICE(REF(C,1),涨停幅);
竞价涨停:=OPEN>=今涨停价;
竞价涨停取反:=NOT(竞价涨停);

{9:25 时 VOL=竞价量（手）}
竞价量股:=VOL*100;
自由流通:=FINANCE(57);
自由流通:=IF(自由流通>0,自由流通,FINANCE(7));
量占自由:=IF(自由流通>0,竞价量股/自由流通,0);

竞价涨幅:=(OPEN/REF(C,1)-1)*100;
均量:=REF(MA(VOL,5),1);
竞价量比:=IF(均量>0,VOL/(均量/240),0);
竞价额:=OPEN*竞价量股;
昨竞价额:=REF(OPEN,1)*REF(VOL,1)*100;
金额比:=IF(昨竞价额>0,竞价额/昨竞价额,0);
竞价换手:=量占自由*100;
流通市值亿:=CAPITAL*REF(C,1)/1000000;

量区间:=竞价量股>1000000 AND 竞价量股<8000000;
涨幅条件:=竞价涨幅>1.5 AND 竞价涨幅<7;
量比条件:=竞价量比>8 AND 竞价量比<35;
金额条件:=金额比>0.9 AND 金额比<6;
换手条件:=竞价换手>0.2 AND 竞价换手<1.5;
市值条件:=流通市值亿>20 AND 流通市值亿<150;

基础池:=主板 AND 非ST AND 昨日涨停 AND 竞价涨停取反;
过滤:=基础池 AND 量区间 AND 涨幅条件 AND 量比条件 AND 金额条件 AND 换手条件 AND 市值条件;

选股:过滤;

{选股设置：按「量占自由」从大到小，取前5。
  高度/炸板次数问财更方便；通达信可用自定义数据或人工复核。}
