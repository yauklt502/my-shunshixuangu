{首板竞价选股 同花顺}
{用法：9:25集合竞价结束后、9:28前做条件选股。结果按「开盘涨跌幅」从小到大排序，取前3只（最多5只）。}
{不要竞价一字涨停：只做竞价低开-8%~-2%。}

创业板:=CODELIKE('30');
涨停幅:=IF(创业板,0.20,0.10);
昨收:=REF(CLOSE,1);
前收:=REF(CLOSE,2);
今涨停价:=ROUND(昨收*(1+涨停幅),2);
今跌停价:=ROUND(昨收*(1-涨停幅),2);
昨涨停价:=ROUND(前收*(1+涨停幅),2);

昨涨停:=REF(CLOSE,1)/前收>=涨停幅*0.95+1 AND REF(CLOSE,1)>=REF(HIGH,1)*0.997;
前日涨停:=REF(CLOSE,2)/REF(CLOSE,3)>=涨停幅*0.95+1 AND REF(CLOSE,2)>=REF(HIGH,2)*0.997;
昨日首板:=昨涨停 AND NOT 前日涨停;
昨日一字:=REF(LOW,1)>=昨涨停价*0.997;
昨日振幅:=REF((HIGH-LOW)/前收,1);

竞价涨幅:=OPEN/昨收-1;
非竞价涨停:=OPEN<今涨停价;
非竞价跌停:=OPEN>今跌停价;
竞价低开区间:=竞价涨幅>=-0.08 AND 竞价涨幅<=-0.02;

昨量比:=REF(VOL,1)/REF(MA(VOL,5),2);
流通市值亿:=FINANCE(7)*昨收/10000;

非ST:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST'));
非科创北交:=NOT(CODELIKE('68') OR CODELIKE('8') OR CODELIKE('4') OR CODELIKE('92'));
大盘竞价:=INDEXO/REF(INDEXC,1)>=0.98;

选股:昨日首板 AND NOT 昨日一字 AND 昨日振幅>=0.015
AND 非竞价涨停 AND 非竞价跌停 AND 竞价低开区间
AND 昨量比>=0.8 AND 昨量比<=5
AND 流通市值亿>=25 AND 流通市值亿<=100
AND 非ST AND 非科创北交 AND 大盘竞价;
