{=========================================================
  晴空买点·条件选股【严格版】
  过滤更严，票更少。先用宽松版跑通后再用本版。
=========================================================}

非停牌:=C>0 AND VOL>0;
非一字:=H<>L;
不追涨:=C/REF(C,1)<1.05;
中期未坏:=C>MA(C,60)*0.90;
过滤:=非停牌 AND 非一字 AND 不追涨 AND 中期未坏;

日跌:=C/REF(C,1)<=0.97;
近5深回:=C/HHV(C,5)<=0.95;
贴日低:=L<H AND (C-L)/(H-L)<=0.35;
B1急跌低吸:=过滤 AND (日跌 OR 近5深回) AND 贴日低 AND VOL<MA(VOL,5)*2.5;

近有跌:=LLV(C,8)/HHV(C,15)<=0.94;
不破前低:=LLV(L,3)>=REF(LLV(L,8),3);
振幅收窄:=(HHV(H,3)-LLV(L,3))/C<=0.08;
B2企稳:=过滤 AND 近有跌 AND 不破前低 AND 振幅收窄 AND VOL<=MA(VOL,5)*1.2 AND C>=MA(C,5);

横盘:=HHV(H,10)/LLV(L,10)<=1.10;
放量:=VOL>MA(VOL,5)*1.5;
转强:=C/REF(C,1)>=1.02 AND C>MA(C,10);
B3启动苗头:=过滤 AND 横盘 AND 放量 AND 转强 AND C>=REF(HHV(H,8),1);

前推升:=REF(C,6)/REF(LLV(C,16),6)>=1.08;
旗内收敛:=(HHV(H,5)-LLV(L,5))/C<=0.08;
突破:=C>REF(HHV(H,5),1) AND C>REF(C,1)*1.015;
B4飘旗突破:=过滤 AND 前推升 AND 旗内收敛 AND 突破 AND VOL>MA(VOL,5)*1.3;

选股:B1急跌低吸 OR B2企稳 OR B3启动苗头 OR B4飘旗突破;
