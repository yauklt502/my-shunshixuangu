{=========================================================
  晴空买点·条件选股【宽松可跑版】
  上一版一个都选不出，多半是过滤叠死了。
  本版去掉：指数牛市框架、60日低位、成交额硬门槛
  只保留：非ST近似、非一字、不追高、四种买点形态

  用法：公式管理器 → 条件选股 → 新建 → 全部粘贴
  周期：日线。选股范围先用「沪深A股」测通，再缩到板块。
  研究用，不构成投资建议。
=========================================================}

{基础过滤（尽量少）}
非停牌:=C>0 AND VOL>0;
非一字:=H<>L;
不追高:=C/REF(C,1)<1.07;
过滤:=非停牌 AND 非一字 AND 不追高;

{===== B1 急跌低吸 / 贴近日低 =====}
{日跌>=2.5% 或 近5日回撤>=4%，收盘靠近当日低点一侧}
日跌:=C/REF(C,1)<=0.975;
近回撤:=C/HHV(C,5)<=0.96;
贴日低:=L<H AND (C-L)/(H-L)<=0.45;
B1:=过滤 AND (日跌 OR 近回撤) AND 贴日低;

{===== B2 企稳（回撤后不再破底 + 站回5日） =====}
近有跌:=LLV(C,10)/HHV(C,20)<=0.95;
不破前低:=LLV(L,3)>=REF(LLV(L,10),3)*0.995;
站回5日:=C>=MA(C,5);
缩量或平量:=VOL<=MA(VOL,5)*1.3;
B2:=过滤 AND 近有跌 AND 不破前低 AND 站回5日 AND 缩量或平量;

{===== B3 启动苗头（横盘后放量转强） =====}
横盘:=HHV(H,10)/LLV(L,10)<=1.12;
放量:=VOL>MA(VOL,5)*1.3;
转强:=C>REF(C,1)*1.015 AND C>MA(C,10);
破上沿:=C>=REF(HHV(H,8),1);
B3:=过滤 AND 横盘 AND 放量 AND 转强 AND 破上沿;

{===== B4 整理后突破 =====}
前有升:=REF(C,5)/REF(LLV(C,15),5)>=1.05;
整理窄:=(HHV(H,5)-LLV(L,5))/C<=0.10;
突破:=C>REF(HHV(H,5),1);
放量突破:=VOL>MA(VOL,5)*1.2;
B4:=过滤 AND 前有升 AND 整理窄 AND 突破 AND 放量突破;

{先跑通：四种任一即可}
选股:B1 OR B2 OR B3 OR B4;

{若仍为空，临时只用下面一行测软件是否正常：}
{选股:C>MA(C,5);}

{只做急跌低吸时改用：}
{选股:B1;}
