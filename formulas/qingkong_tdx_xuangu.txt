{=========================================================
  晴空买点·条件选股【实用版】
  宽松版会出两千多只=没选；严格版又易全灭。
  本版默认只跑 B1（急跌低吸），目标大致几十到一两百只。
  研究用，不构成投资建议。

  用法：条件选股 → 新建 → 全部粘贴
  范围：沪深A股；周期：日线
  先看 B1；要企稳/启动/飘旗时，改最后一行
=========================================================}

{----- 基础过滤：去掉垃圾、去掉追高、只要还在中期框架里 -----}
非停牌:=C>0 AND VOL>0;
非一字:=H<>L;
不追高:=C/REF(C,1)<1.03;
中期未坏:=C>MA(C,60)*0.93;
未高潮:=C/HHV(C,60)<0.97;
{CAPITAL=流通股本(手)。换手过低的冷门股不做}
换手够:=IF(CAPITAL>0,VOL/CAPITAL*100>1,VOL>MA(VOL,5));
过滤:=非停牌 AND 非一字 AND 不追高 AND 中期未坏 AND 未高潮 AND 换手够;

{===== B1 急跌低吸（默认）=====}
{日跌>=3.5%，或近5日从高点回撤>=6%；收在当日偏低位置；排除天量崩盘}
日急跌:=C/REF(C,1)<=0.965;
近深回:=C/HHV(C,5)<=0.94;
贴日低:=(C-L)/(H-L+0.0001)<=0.30;
非崩盘量:=VOL<MA(VOL,5)*2.8;
B1:=过滤 AND (日急跌 OR 近深回) AND 贴日低 AND 非崩盘量;

{===== B2 企稳（更严，单独用）=====}
近有跌:=LLV(C,10)/HHV(C,20)<=0.93;
不破底:=LLV(L,3)>=REF(LLV(L,10),3);
收窄:=(HHV(H,3)-LLV(L,3))/C<=0.06;
缩量:=VOL<=MA(VOL,5);
站上5日:=C>=MA(C,5) AND C>REF(C,1);
B2:=过滤 AND 近有跌 AND 不破底 AND 收窄 AND 缩量 AND 站上5日;

{===== B3 启动苗头（更严，单独用）=====}
横盘:=HHV(H,12)/LLV(L,12)<=1.08;
放量:=VOL>MA(VOL,5)*1.8 AND VOL>REF(VOL,1)*1.3;
转强:=C/REF(C,1)>=1.025 AND C>MA(C,10) AND C>MA(C,20);
破上沿:=C>REF(HHV(H,10),1);
B3:=过滤 AND 横盘 AND 放量 AND 转强 AND 破上沿;

{===== B4 飘旗突破（更严，单独用）=====}
前推升:=REF(C,6)/REF(LLV(C,18),6)>=1.10;
旗收敛:=(HHV(H,6)-LLV(L,6))/C<=0.07;
突破:=C>REF(HHV(H,6),1) AND C/REF(C,1)>=1.02;
放量:=VOL>MA(VOL,5)*1.5;
B4:=过滤 AND 前推升 AND 旗收敛 AND 突破 AND 放量;

{默认只做急跌低吸。不要 B1 OR B2 OR B3 OR B4，会再次上千只}
选股:B1;

{改用其它买点时，只保留一行：}
{选股:B2;}
{选股:B3;}
{选股:B4;}

{B1 仍超过300只：把日急跌改成 0.96（跌4%），或换手够改成 >1.5}
{B1 仍少于10只：把日急跌改成 0.975，近深回改成 0.96}
