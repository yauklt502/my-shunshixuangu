{=========================================================
  晴空买点·仅B1急跌低吸【最简】
  若实用版因 CAPITAL 报错，用本公式（不用流通股本）
=========================================================}

非停牌:=C>0 AND VOL>0 AND H<>L;
不追高:=C/REF(C,1)<1.03;
中期:=C>MA(C,60)*0.93;
未高潮:=C/HHV(C,60)<0.97;
量能:=VOL>MA(VOL,20)*0.8 AND VOL>REF(MA(VOL,5),1)*0.6;

日急跌:=C/REF(C,1)<=0.965;
近深回:=C/HHV(C,5)<=0.94;
贴日低:=(C-L)/(H-L+0.0001)<=0.30;
非崩:=VOL<MA(VOL,5)*2.8;

选股:非停牌 AND 不追高 AND 中期 AND 未高潮 AND 量能 AND (日急跌 OR 近深回) AND 贴日低 AND 非崩;
