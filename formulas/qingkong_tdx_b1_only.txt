{=========================================================
  晴空·B1急跌低吸【收敛版·无CAPITAL】
  CAPITAL报错时用这个
=========================================================}

非停牌:=C>0 AND VOL>0 AND H>L;
不追高:=C/REF(C,1)<1.02;
多头:=MA(C,20)>MA(C,60) AND C>MA(C,60)*0.97;
未高潮:=C/HHV(C,60)<0.96;
量能:=VOL>MA(VOL,20) AND VOL>MA(VOL,5)*0.9;

日急跌:=C/REF(C,1)<=0.96;
贴日低:=(C-L)/(H-L)<=0.25;
回踩10:=L<=MA(C,10)*1.02 AND L>=MA(C,10)*0.97;
回踩20:=L<=MA(C,20)*1.02 AND L>=MA(C,20)*0.97;
非崩盘:=VOL<MA(VOL,5)*2.5 AND C>MA(C,60)*0.95;

选股:非停牌 AND 不追高 AND 多头 AND 未高潮 AND 量能 AND 日急跌 AND 贴日低 AND (回踩10 OR 回踩20) AND 非崩盘;
