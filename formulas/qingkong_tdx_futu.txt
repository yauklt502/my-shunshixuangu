{=========================================================
  晴空买点·副图打点（通达信）
  与选股公式同一套 B1–B4，画在副图核对形态
  公式管理器 → 技术指标 → 新建 → 副图
=========================================================}

非ST:=NOT(NAMELIKE('ST'));
非一字:=H<>L AND NOT(O=C AND C=H AND C>=REF(C,1)*1.09);
不追涨:=C/REF(C,1)<1.05;
牛市框架:=INDEXC>MA(INDEXC,60);
中期未坏:=C>MA(C,60)*0.92 AND MA(C,60)>REF(MA(C,60),3);
低位未高潮:=C/HHV(C,60)<0.92;
过滤:=非ST AND 非一字 AND 不追涨 AND 牛市框架 AND 中期未坏 AND 低位未高潮 AND VOL>0;

日跌:=C/REF(C,1)<=0.97;
近5深回:=C/HHV(C,5)<=0.95;
贴日低:=IF(H>L,(C-L)/(H-L)<=0.35,0);
非天量崩:=VOL<MA(VOL,5)*2.2;
B1:=过滤 AND (日跌 OR 近5深回) AND 贴日低 AND 非天量崩;

近有跌:=LLV(C,8)/HHV(C,15)<=0.94;
不破前低:=LLV(L,3)>=REF(LLV(L,8),3);
振幅收窄:=(HHV(H,3)-LLV(L,3))/C<=0.07;
缩量:=VOL<=MA(VOL,5)*1.1;
B2:=过滤 AND 近有跌 AND 不破前低 AND 振幅收窄 AND 缩量 AND C>=MA(C,5);

横盘:=COUNT(ABS(C-REF(C,1))/REF(C,1)<0.025,10)>=6;
放量:=VOL>MA(VOL,5)*1.5 AND VOL>REF(VOL,1)*1.2;
转强:=C/REF(C,1)>=1.02 AND C>MA(C,10);
B3:=过滤 AND 横盘 AND 放量 AND 转强 AND C>=REF(HHV(H,8),1);

前推升:=REF(C,6)/REF(LLV(C,16),6)>=1.08;
旗内收敛:=(HHV(H,5)-LLV(L,5))/MA(C,5)<=0.08;
旗内不过热:=HHV(C,5)/LLV(C,5)<1.10;
突破:=C>REF(HHV(H,5),1) AND C>REF(C,1)*1.015;
B4:=过滤 AND 前推升 AND 旗内收敛 AND 旗内不过热 AND 突破 AND VOL>MA(VOL,5)*1.3;

STICKLINE(B1,1,2,8,0),COLORRED;
STICKLINE(B2,1,2,8,0),COLORGREEN;
STICKLINE(B3,1,2,8,0),COLORYELLOW;
STICKLINE(B4,1,2,8,0),COLORBLUE;
DRAWTEXT(B1,2.2,'B1低吸'),COLORRED;
DRAWTEXT(B2,2.2,'B2企稳'),COLORGREEN;
DRAWTEXT(B3,2.2,'B3启动'),COLORYELLOW;
DRAWTEXT(B4,2.2,'B4飘旗'),COLORBLUE;
