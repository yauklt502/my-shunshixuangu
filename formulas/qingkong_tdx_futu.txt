{=========================================================
  晴空·副图打点（与实用版选股同一阈值）
=========================================================}

非停牌:=C>0 AND VOL>0;
非一字:=H<>L;
不追高:=C/REF(C,1)<1.03;
中期未坏:=C>MA(C,60)*0.93;
未高潮:=C/HHV(C,60)<0.97;
换手够:=IF(CAPITAL>0,VOL/CAPITAL*100>1,VOL>MA(VOL,5));
过滤:=非停牌 AND 非一字 AND 不追高 AND 中期未坏 AND 未高潮 AND 换手够;

日急跌:=C/REF(C,1)<=0.965;
近深回:=C/HHV(C,5)<=0.94;
贴日低:=(C-L)/(H-L+0.0001)<=0.30;
B1:=过滤 AND (日急跌 OR 近深回) AND 贴日低 AND VOL<MA(VOL,5)*2.8;

近有跌:=LLV(C,10)/HHV(C,20)<=0.93;
不破底:=LLV(L,3)>=REF(LLV(L,10),3);
B2:=过滤 AND 近有跌 AND 不破底 AND (HHV(H,3)-LLV(L,3))/C<=0.06 AND VOL<=MA(VOL,5) AND C>=MA(C,5) AND C>REF(C,1);

横盘:=HHV(H,12)/LLV(L,12)<=1.08;
B3:=过滤 AND 横盘 AND VOL>MA(VOL,5)*1.8 AND C/REF(C,1)>=1.025 AND C>MA(C,20) AND C>REF(HHV(H,10),1);

前推升:=REF(C,6)/REF(LLV(C,18),6)>=1.10;
B4:=过滤 AND 前推升 AND (HHV(H,6)-LLV(L,6))/C<=0.07 AND C>REF(HHV(H,6),1) AND C/REF(C,1)>=1.02 AND VOL>MA(VOL,5)*1.5;

STICKLINE(B1,1,2,8,0),COLORRED;
STICKLINE(B2,1,2,8,0),COLORGREEN;
STICKLINE(B3,1,2,8,0),COLORYELLOW;
STICKLINE(B4,1,2,8,0),COLORBLUE;
DRAWTEXT(B1,2.2,'B1'),COLORRED;
DRAWTEXT(B2,2.2,'B2'),COLORGREEN;
DRAWTEXT(B3,2.2,'B3'),COLORYELLOW;
DRAWTEXT(B4,2.2,'B4'),COLORBLUE;
