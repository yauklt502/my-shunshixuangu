{=========================================================
  晴空·副图打点（与宽松选股同一套）
=========================================================}

非停牌:=C>0 AND VOL>0;
非一字:=H<>L;
不追高:=C/REF(C,1)<1.07;
过滤:=非停牌 AND 非一字 AND 不追高;

日跌:=C/REF(C,1)<=0.975;
近回撤:=C/HHV(C,5)<=0.96;
贴日低:=L<H AND (C-L)/(H-L)<=0.45;
B1:=过滤 AND (日跌 OR 近回撤) AND 贴日低;

近有跌:=LLV(C,10)/HHV(C,20)<=0.95;
不破前低:=LLV(L,3)>=REF(LLV(L,10),3)*0.995;
B2:=过滤 AND 近有跌 AND 不破前低 AND C>=MA(C,5) AND VOL<=MA(VOL,5)*1.3;

横盘:=HHV(H,10)/LLV(L,10)<=1.12;
放量:=VOL>MA(VOL,5)*1.3;
转强:=C>REF(C,1)*1.015 AND C>MA(C,10);
B3:=过滤 AND 横盘 AND 放量 AND 转强 AND C>=REF(HHV(H,8),1);

前有升:=REF(C,5)/REF(LLV(C,15),5)>=1.05;
整理窄:=(HHV(H,5)-LLV(L,5))/C<=0.10;
B4:=过滤 AND 前有升 AND 整理窄 AND C>REF(HHV(H,5),1) AND VOL>MA(VOL,5)*1.2;

STICKLINE(B1,1,2,8,0),COLORRED;
STICKLINE(B2,1,2,8,0),COLORGREEN;
STICKLINE(B3,1,2,8,0),COLORYELLOW;
STICKLINE(B4,1,2,8,0),COLORBLUE;
DRAWTEXT(B1,2.2,'B1'),COLORRED;
DRAWTEXT(B2,2.2,'B2'),COLORGREEN;
DRAWTEXT(B3,2.2,'B3'),COLORYELLOW;
DRAWTEXT(B4,2.2,'B4'),COLORBLUE;
