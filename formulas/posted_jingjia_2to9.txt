创业板:=CODELIKE('30');
涨停幅:=IF(创业板,0.20,0.10);
昨收:=REF(C,1);
今涨停价:=ROUND(昨收*(1+涨停幅),2);
竞价涨幅:=OPEN/昨收-1;
昨量比:=REF(V,1)/REF(MA(V,5),2);
流通市值亿:=CAPITAL*昨收/1000000;
非ST:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST'));

选股:非ST
AND 竞价涨幅>=0.02 AND 竞价涨幅<=0.09
AND OPEN<今涨停价
AND 昨收>=11
AND 流通市值亿>=25
AND 昨量比>=0.7;
